<script>
  (function () {
    var data = JSON.parse(document.getElementById('johnisms-data').textContent);
    var list = document.getElementById('random-johnisms-list');
    var shuffleLink = document.getElementById('reshuffle-johnisms');

    function pickTen() {
      var pool = data.slice();
      var picks = [];
      for (var i = 0; i < 10 && pool.length > 0; i++) {
        var idx = Math.floor(Math.random() * pool.length);
        picks.push(pool[idx]);
        pool.splice(idx, 1);
      }
      return picks;
    }

    function render() {
      var picks = pickTen();
      list.innerHTML = '';
      picks.forEach(function (text) {
        var li = document.createElement('li');
        li.style.marginBottom = '0.75rem';
        li.style.fontStyle = 'italic';
        li.textContent = '\u201C' + text + '\u201D';
        list.appendChild(li);
      });
    }

    render();

    if (shuffleLink) {
      shuffleLink.addEventListener('click', function (e) {
        e.preventDefault();
        render();
      });
    }
  })();
</script>
